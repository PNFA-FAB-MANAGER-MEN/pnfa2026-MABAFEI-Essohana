```python


import machine
import time
from machine import Pin

ir_sensor = Pin(2, Pin.IN)
moteur = Pin(3, Pin.OUT)
moteur.value(0)

# 1. Configuration de la broche GPIO et du PWM
servo_pin = machine.Pin(1)
servo = machine.PWM(servo_pin, freq=50)

def set_angle(angle):
    """
    Calcule le cycle de travail (duty cycle) pour obtenir l'angle souhaité.
    Pour un servomoteur standard (0 à 180°), l'impulsion dure entre 0.5ms et 2.5ms.
    En résolution 10 bits (0-1023) à 50Hz : 0.5ms vaut ~26, 2.5ms vaut ~123.
    """
    duty = int(26 + (angle / 180) * (123 - 26))
    servo.duty(duty)

def marquage():
    print("Position : 0 degré")
    set_angle(0)
    time.sleep(2)

    print("Position : 180 degrés")
    set_angle(180)
    time.sleep(2)

# 2. Boucle principale de test
print("Démarrage du contrôle du servomoteur...")

try:
    while True:
        valeurpin = ir_sensor.value()

        if valeurpin == 0:
            moteur.value(1)
            time.sleep(1)
            marquage()
            moteur.value(0)
            time.sleep(2)

except KeyboardInterrupt:
    # Arrête proprement le signal PWM si vous cliquez sur "Stop" dans Thonny
    servo.deinit()
    print("Programme arrêté.")
```
